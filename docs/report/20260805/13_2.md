```bash

npm run dev -- --hostname 127.0.0.1 --port 3000

# 관측 재료
POST /api/projects → /api/projects/{id}/repositories → /api/repository-sets/{id}/sync

cd frontend && npx tsc --noEmit && npx playwright test e2e/flow-canvas.spec.ts && npm run build
```
